\begin{lemma}\label{lem:parts}
Every real $u\in C^2([0,1])$ with $u(0)=u(1)=0$ satisfies
$\int_0^1 u(x)u''(x)\,dx=-\int_0^1 |u'(x)|^2\,dx$.
\end{lemma}
\begin{proof}
This follows from integration by parts.
\end{proof}
